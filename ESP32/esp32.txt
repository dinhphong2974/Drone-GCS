import machine
import network
import socket
import time
import struct
import gc

# ══════════════════════════════════════════════════════════════
# 0. DEBUG LOG (Tắt khi bay thật để giảm latency)
# ══════════════════════════════════════════════════════════════
DEBUG = True


def _log(msg):
    if DEBUG:
        print(msg)


# ══════════════════════════════════════════════════════════════
# 1. UART CONFIG (ESP32 UART1 <-> FC UART3)
# ══════════════════════════════════════════════════════════════
uart = machine.UART(1, baudrate=115200, tx=17, rx=16, txbuf=512, rxbuf=512)

# ══════════════════════════════════════════════════════════════
# 2. WIFI AP CONFIG
# ══════════════════════════════════════════════════════════════
ap = network.WLAN(network.AP_IF)
ap.active(True)
ap.config(essid="Drone_GCS_Wifi", password="DroneGCS@2026!", authmode=4)

_log("Đang khởi tạo Wifi...")
while not ap.active():
    time.sleep_ms(100)
_log("Wifi AP đã sẵn sàng. IP: " + ap.ifconfig()[0])

# ══════════════════════════════════════════════════════════════
# 3. TCP SERVER
# ══════════════════════════════════════════════════════════════
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
s.bind(('', 8080))
s.listen(1)
s.setblocking(False)

_log("Đang chờ PC kết nối tới cổng 8080...")

# ══════════════════════════════════════════════════════════════
# 4. WATCHDOG (tự reset nếu treo)
# ══════════════════════════════════════════════════════════════
try:
    from machine import WDT
    wdt = WDT(timeout=5000)
    _log("[WDT] Watchdog Timer đã bật — timeout 5s")
except Exception:
    wdt = None
    _log("[WDT] Không hỗ trợ WDT trên board này")

# ══════════════════════════════════════════════════════════════
# 5. FAILSAFE & RC CONSTANTS
# ══════════════════════════════════════════════════════════════
FAILSAFE_TIMEOUT_MS = 2000
FAILSAFE_RC_INTERVAL_MS = 100

RC_CENTER = 1500
RC_LOW = 1000
RC_HIGH = 2000

CH_ROLL = 0
CH_PITCH = 1
CH_THROTTLE = 2
CH_YAW = 3
CH_AUX1 = 4
CH_AUX2 = 5
CH_AUX3 = 6
CH_AUX4 = 7

MSP_SET_RAW_RC = 200

FS_PREFIX = b'FS:'
EMERGENCY_PREFIX = b'EM:'
HEARTBEAT_PREFIX = b'HB:'
PING_PREFIX = b'PING:'

MARKERS = (EMERGENCY_PREFIX, FS_PREFIX, HEARTBEAT_PREFIX, PING_PREFIX, b'$M<')

conn = None
last_msg_time = time.ticks_ms()
failsafe_triggered = False
last_failsafe_rc_time = 0
conn_error_count = 0

failsafe_behavior = "rth"

gc_counter = 0
GC_INTERVAL = 200

CONN_ERROR_THRESHOLD = 10

# ══════════════════════════════════════════════════════════════
# 6. MSP FRAME (ZERO ALLOC)
# ══════════════════════════════════════════════════════════════
_rc_frame_buf = bytearray(22)
_rc_frame_buf[0:3] = b'$M<'
_rc_frame_buf[3] = 16
_rc_frame_buf[4] = MSP_SET_RAW_RC

_FAILSAFE_RTH_CHANNELS = (
    RC_CENTER, RC_CENTER, RC_CENTER, RC_CENTER,
    RC_HIGH, RC_CENTER, RC_LOW, RC_HIGH
)

_FAILSAFE_LAND_CHANNELS = (
    RC_CENTER, RC_CENTER, RC_CENTER, RC_CENTER,
    RC_HIGH, RC_CENTER, RC_HIGH, RC_LOW
)


def build_msp_set_raw_rc(channels):
    struct.pack_into('<8H', _rc_frame_buf, 5, *channels)
    checksum = 16 ^ MSP_SET_RAW_RC
    for i in range(5, 21):
        checksum ^= _rc_frame_buf[i]
    _rc_frame_buf[21] = checksum & 0xFF
    return _rc_frame_buf


def send_failsafe_rth():
    frame = build_msp_set_raw_rc(_FAILSAFE_RTH_CHANNELS)
    uart.write(frame)


def send_failsafe_land():
    frame = build_msp_set_raw_rc(_FAILSAFE_LAND_CHANNELS)
    uart.write(frame)


def check_failsafe_config(data):
    global failsafe_behavior
    if data.startswith(FS_PREFIX):
        config = data[len(FS_PREFIX):].strip()
        if config == b'rth':
            failsafe_behavior = "rth"
            _log("[CONFIG] Failsafe: RTH khi mất WiFi")
        elif config == b'ignore':
            failsafe_behavior = "ignore"
            _log("[CONFIG] Failsafe: IGNORE khi mất WiFi")
        else:
            _log("[CONFIG] Không nhận dạng: " + str(config))
        return True
    return False


def _find_next_marker(data, start=1):
    idxs = []
    for marker in MARKERS:
        idx = data.find(marker, start)
        if idx >= 0:
            idxs.append(idx)
    return min(idxs) if idxs else -1


def _split_segment(data):
    next_idx = _find_next_marker(data, 1)
    if next_idx == -1:
        return data, b''
    return data[:next_idx], data[next_idx:]


def _contains_rc_command(chunk):
    pos = 0
    while True:
        idx = chunk.find(b'$M<', pos)
        if idx < 0:
            return False
        if idx + 5 < len(chunk) and chunk[idx + 4] == MSP_SET_RAW_RC:
            return True
        pos = idx + 3


def _handle_emergency_chain(data):
    global failsafe_triggered
    frame_count = 0
    remaining = data

    if not remaining.startswith(EMERGENCY_PREFIX):
        return remaining, frame_count

    uart.read()

    while remaining.startswith(EMERGENCY_PREFIX):
        remaining = remaining[len(EMERGENCY_PREFIX):]
        next_idx = _find_next_marker(remaining, 0)
        if next_idx == -1:
            msp_frame = remaining
            remaining = b''
        else:
            msp_frame = remaining[:next_idx]
            remaining = remaining[next_idx:]

        if msp_frame:
            uart.write(msp_frame)
            frame_count += 1

    if failsafe_triggered:
        failsafe_triggered = False
        _log("[!] EMERGENCY — Failsafe bị tắt bởi lệnh khẩn cấp")

    return remaining, frame_count


def close_connection():
    global conn, conn_error_count
    if conn is not None:
        try:
            conn.close()
        except:
            pass
        conn = None
        conn_error_count = 0
        _log("[!] TCP connection closed")


# ══════════════════════════════════════════════════════════════
# 7. MAIN LOOP
# ══════════════════════════════════════════════════════════════
gc.collect()
_log("[MEM] Free: " + str(gc.mem_free()) + " bytes")

while True:
    if wdt:
        wdt.feed()

    if conn is None:
        if failsafe_triggered and failsafe_behavior == "rth":
            now = time.ticks_ms()
            if time.ticks_diff(now, last_failsafe_rc_time) > FAILSAFE_RC_INTERVAL_MS:
                send_failsafe_rth()
                last_failsafe_rc_time = now

        if uart.any():
            uart.read()

        try:
            conn, addr = s.accept()
            conn.setblocking(False)
            _log("\n[+] PC đã kết nối từ: " + str(addr))
            last_msg_time = time.ticks_ms()
            failsafe_triggered = False
            conn_error_count = 0
        except OSError:
            pass

        time.sleep_ms(10)

    else:
        try:
            data_from_pc = conn.recv(1024)

            if data_from_pc == b'':
                print("[!] PC đã ngắt kết nối (empty recv)")
                close_connection()
                continue

            if data_from_pc:
                conn_error_count = 0

                data = data_from_pc
                while data:
                    marker_idx = _find_next_marker(data, 0)

                    if marker_idx == -1 or marker_idx > 0:
                        segment = data if marker_idx == -1 else data[:marker_idx]
                        uart.write(segment)
                        last_msg_time = time.ticks_ms()

                        if _contains_rc_command(segment):
                            try:
                                conn.send(b'ACK:RC\n')
                            except:
                                pass

                        data = b'' if marker_idx == -1 else data[marker_idx:]
                        continue

                    if data.startswith(EMERGENCY_PREFIX):
                        data, frame_count = _handle_emergency_chain(data)
                        last_msg_time = time.ticks_ms()
                        try:
                            conn.send(b'ACK:EM\n')
                        except:
                            pass
                        continue

                    if data.startswith(FS_PREFIX):
                        segment, data = _split_segment(data)
                        if check_failsafe_config(segment):
                            last_msg_time = time.ticks_ms()
                            try:
                                conn.send(b'ACK:FS\n')
                            except:
                                pass
                        continue

                    if data.startswith(PING_PREFIX):
                        segment, data = _split_segment(data)
                        last_msg_time = time.ticks_ms()
                        try:
                            pong_reply = b'PONG:' + segment[len(PING_PREFIX):] + b'\n'
                            conn.send(pong_reply)
                        except:
                            pass
                        continue

                    if data.startswith(HEARTBEAT_PREFIX):
                        _, data = _split_segment(data)
                        last_msg_time = time.ticks_ms()
                        continue

                    uart.write(data)
                    last_msg_time = time.ticks_ms()
                    if _contains_rc_command(data):
                        try:
                            conn.send(b'ACK:RC\n')
                        except:
                            pass
                    data = b''

                if failsafe_triggered:
                    _log("[+] PC đã kết nối lại — tắt Failsafe")
                    failsafe_triggered = False

        except OSError:
            pass

        if uart.any():
            data_from_fc = uart.read(256)
            if data_from_fc and conn is not None:
                try:
                    conn.send(data_from_fc)
                    conn_error_count = 0
                except OSError:
                    conn_error_count += 1
                    if conn_error_count >= CONN_ERROR_THRESHOLD:
                        _log("[!] Send lỗi liên tiếp — đóng kết nối")
                        close_connection()
                        continue

        elapsed = time.ticks_diff(time.ticks_ms(), last_msg_time)

        if elapsed > FAILSAFE_TIMEOUT_MS:
            if not failsafe_triggered:
                failsafe_triggered = True
                _log("\n[!] MẤT KẾT NỐI VỚI PC!")

                if failsafe_behavior == "rth":
                    _log("[FAILSAFE] Kích hoạt RTH — Drone bay về Home")
                    send_failsafe_rth()
                elif failsafe_behavior == "ignore":
                    _log("[FAILSAFE] IGNORE — Drone tiếp tục bay mission")
                else:
                    _log("[FAILSAFE] Mặc định: Kích hoạt RTH")
                    send_failsafe_rth()

                last_failsafe_rc_time = time.ticks_ms()
            else:
                now = time.ticks_ms()
                if time.ticks_diff(now, last_failsafe_rc_time) > FAILSAFE_RC_INTERVAL_MS:
                    if failsafe_behavior == "rth":
                        send_failsafe_rth()
                    last_failsafe_rc_time = now

    time.sleep_ms(1)
    gc_counter += 1
    if gc_counter >= GC_INTERVAL:
        gc.collect()
        gc_counter = 0
